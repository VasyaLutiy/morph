import { findCard } from "./cardBrief.js";
import type { DebtResult } from "./cardBrief.js";
import { DEFAULT_TIMEOUT_MS, runAcceptance } from "../acceptance/run.js";
import { gitOk } from "../git/run.js";
import { commitPaths } from "../git/commit.js";
import type { Diffstat, Trailer } from "../git/types.js";

export interface AcceptOptions {
  deck: string;
  id: string;
  model: string;
  commit: boolean;
}

export interface AcceptDeps {
  env: Record<string, string>;
  timeoutMs?: number;
}

export interface AcceptDocument {
  deck: string;
  card: string;
  model: string;
  exit: number | null;
  timedOut: boolean;
  green: boolean;
  log: string;
  outside: string[] | null;
  commit: string | null;
  diffstat: Diffstat | null;
  reason: string | null;
}

/**
 * The paths under `git status --porcelain --untracked-files=all` that are
 * neither a target nor under `.morph/`: the tree the payer left, taken before
 * the acceptance runs. The paths are the text after the two status letters and
 * a space (after " -> " for a rename), distinct and sorted by code unit. Throws
 * (through gitOk) outside a repository.
 */
export function changedOutside(
  root: string,
  targets: string[],
  env: Record<string, string>,
): string[] {
  const status = gitOk(root, ["status", "--porcelain", "--untracked-files=all"], env);
  const targetSet = new Set(targets);
  const found = new Set<string>();
  for (const line of status.split("\n")) {
    if (line === "") {
      continue;
    }
    let entry = line.length > 3 ? line.slice(3) : "";
    const arrow = entry.indexOf(" -> ");
    if (arrow !== -1) {
      entry = entry.slice(arrow + 4);
    }
    if (entry === "" || entry.startsWith(".morph/") || targetSet.has(entry)) {
      continue;
    }
    found.add(entry);
  }
  return [...found].sort((a, b) => (a < b ? -1 : a > b ? 1 : 0));
}

export async function acceptCard(
  root: string,
  args: AcceptOptions,
  deps: AcceptDeps,
): Promise<DebtResult> {
  const found = findCard(root, args.deck, args.id);
  if (!found.ok) {
    return found.result;
  }
  const card = found.card;

  if (card.acceptance === null || card.acceptance.trim() === "") {
    return {
      code: 2,
      document: {
        error: {
          code: 2,
          kind: "RefusalError",
          message: `card ${card.customId} has no acceptance`,
        },
      },
    };
  }

  const outside = args.commit ? changedOutside(root, card.targets, deps.env) : null;

  const result = await runAcceptance(card.acceptance, root, {
    env: deps.env,
    timeoutMs: deps.timeoutMs ?? DEFAULT_TIMEOUT_MS,
  });

  const green = result.exit === 0;

  const reasons: string[] = [];
  if (!green) {
    reasons.push(`acceptance failed (exit ${String(result.exit)})`);
  }
  if (outside !== null && outside.length > 0) {
    reasons.push(`changed outside the targets: ${outside.join(", ")}`);
  }

  let commit: string | null = null;
  let diffstat: Diffstat | null = null;
  if (args.commit && reasons.length === 0) {
    const subject = `morph ${card.customId}: ${card.targets.join(", ")}`;
    const trailers: Trailer[] = [
      ["Morph-Card", card.customId],
      ["Morph-Model", args.model],
      ["Morph-Acceptance-Exit", "0"],
      ["Morph-Debt", "true"],
    ];
    const info = commitPaths(root, card.targets, subject, trailers, deps.env, true);
    if (info === null) {
      reasons.push("nothing to commit: the targets equal HEAD");
    } else {
      commit = info.commit;
      diffstat = info.diffstat;
    }
  }

  const document: AcceptDocument = {
    deck: args.deck,
    card: card.customId,
    model: args.model,
    exit: result.exit,
    timedOut: result.timedOut,
    green,
    log: result.log,
    outside,
    commit,
    diffstat,
    reason: reasons.length === 0 ? null : reasons.join("; "),
  };

  const ok = green && (!args.commit || commit !== null);
  return { code: ok ? 0 : 1, document };
}
