function at(row: string[], index: number): string {
  if (index < 0 || index >= row.length) return "";
  const value = row[index];
  return value === undefined ? "" : value;
}

function cells(line: string): string[] {
  let text = line.trim();
  if (text.startsWith("|")) text = text.slice(1);
  if (text.endsWith("|") && !text.endsWith("\\|")) text = text.slice(0, -1);
  return text
    .split(/(?<!\\)\|/)
    .map((part) => part.trim().split("\\|").join("|"));
}

function bare(text: string): string {
  return text.split("**").join("").trim();
}

function tablesOf(text: string | null): Table[] {
  if (text === null) return [];
  const lines = text.split("\n");
  const tables: Table[] = [];
  let index = 0;
  while (index < lines.length) {
    const head = at(lines, index).trim();
    const next = at(lines, index + 1).trim();
    if (head.startsWith("|") && index + 1 < lines.length && SEPARATOR.test(next)) {
      const rows: string[][] = [];
      let row = index + 2;
      while (row < lines.length && at(lines, row).trim().startsWith("|")) {
        rows.push(cells(at(lines, row)));
        row += 1;
      }
      tables.push({ header: cells(at(lines, index)), rows });
      index = row;
    } else {
      index += 1;
    }
  }
  return tables;
}

function columnIndex(header: string[], match: (name: string) => boolean): number {
  for (let index = 0; index < header.length; index += 1) {
    if (match(bare(at(header, index)))) return index;
  }
  return -1;
}

function cutTail(text: string, limit: number): string {
  return text.length > limit ? text.slice(0, limit - 1) + "…" : text;
}

function dateOf(id: string): string {
  const head = id.slice(0, 8);
  return head.slice(0, 4) + "-" + head.slice(4, 6) + "-" + head.slice(6, 8);
}

function modelsOf(ids: string[], runs: StoryRun[]): string[] {
  const models: string[] = [];
  for (const id of ids) {
    const run = runs.find((candidate) => candidate.runId === id);
    if (run === undefined) continue;
    for (const model of run.models) {
      if (!models.includes(model)) models.push(model);
    }
  }
  return models;
}
