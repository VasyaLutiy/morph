/**
 * Where an acceptance log stopped, and its compile lines that name a file
 * outside the card's targets. Pure: no import at all, no Node module.
 */

export interface StubVerdict {
  stage: string | null;
  expected: string;
  outside: string[];
  failures: string[];
}

/**
 * The stage at which a card's own run is expected to stop: "probe" when the
 * acceptance echoes `== probe`, else "guard" (a code card is red at its probe,
 * a judge before its file at its guard).
 */
export function expectedStage(acceptance: string): string {
  for (const line of acceptance.split("\n")) {
    if (line.startsWith("echo '== probe'")) {
      return "probe";
    }
  }
  return "guard";
}

/**
 * Read one acceptance log: the stage it stopped at, the lines of its compile
 * stages that name a file outside the card's targets, and the failures at the
 * expected stage. `stages` null means every stage. Lists keep first-seen order.
 */
export function stubVerdict(
  log: string,
  expected: string,
  targets: readonly string[],
  fileLines: readonly string[],
  stages: readonly string[] | null = ["build", "vet", "tsc"],
): StubVerdict {
  const outside: string[] = [];
  const failures: string[] = [];
  let current: string | null = null;

  for (const raw of log.split("\n")) {
    const line = raw.trimEnd();

    const header = /^== (\S+)/.exec(line);
    if (header !== null) {
      current = line.includes(" failed (see above)") ? "verdict" : header[1];
      continue;
    }

    if (current === null) {
      continue;
    }

    const t = line.trim();
    if (current === expected && (t.startsWith("FAIL ") || t.startsWith("--- FAIL: "))) {
      if (!failures.includes(t)) {
        failures.push(t);
      }
    }

    if (
      (stages === null || stages.includes(current)) &&
      line !== "" &&
      !line.startsWith(" ") &&
      !line.startsWith("\t")
    ) {
      for (const pattern of fileLines) {
        const match = new RegExp(pattern).exec(line);
        if (match !== null) {
          if (!targets.includes(match[1]) && !outside.includes(line)) {
            outside.push(line);
          }
          break;
        }
      }
    }
  }

  return { stage: current, expected, outside, failures };
}
