import fs from "node:fs";
import os from "node:os";
import path from "node:path";
import { runAcceptance } from "../acceptance/run.js";
import { gitOk } from "../git/run.js";
import { TREE_PROFILES } from "../language/treeProfiles.js";
import { expectedStage, stubVerdict } from "./stubVerdict.js";
import type { Card } from "../cards/types.js";
import type { GatePhase, GateStep } from "./gatePlan.js";

export interface PlayRow {
  phase: GatePhase;
  generation: number;
  card: string;
  exit: number | null;
  timedOut: boolean;
  seconds: number;
  stage: string | null;
  expected: string | null;
  outside: string[];
  failures: string[];
  ok: boolean;
}

export interface PlayInput {
  root: string;
  cards: readonly Card[];
  steps: readonly GateStep[];
  stubDir: string;
  refDir: string;
}

export interface PlayDeps {
  env: Record<string, string>;
  now: () => number;
  timeoutMs?: number;
  before?: (scratch: string) => void;
}

/**
 * Plays a gate plan in a scratch clone of root's committed HEAD: one row per
 * acceptance run. The clone borrows root's objects (`--shared`) and is checked
 * out detached at HEAD, so the working tree of root is never read and never
 * written; the scratch base is removed in a finally.
 */
export async function playGate(input: PlayInput, deps: PlayDeps): Promise<PlayRow[]> {
  const fileLines = TREE_PROFILES.map((p) => p.fileLine);

  const byId = new Map<string, Card>();
  for (const card of input.cards) {
    byId.set(card.customId, card);
  }

  const base = fs.mkdtempSync(path.join(os.tmpdir(), "morph-gate-"));
  try {
    const scratch = path.join(base, "w");

    gitOk(
      input.root,
      ["clone", "-q", "--shared", "--no-checkout", input.root, scratch],
      deps.env,
    );
    const head = gitOk(input.root, ["rev-parse", "HEAD"], deps.env).trim();
    gitOk(scratch, ["checkout", "-q", "--detach", head], deps.env);

    const nodeModules = path.join(input.root, "node_modules");
    if (fs.existsSync(nodeModules)) {
      fs.symlinkSync(fs.realpathSync(nodeModules), path.join(scratch, "node_modules"));
      fs.mkdirSync(path.join(scratch, ".git", "info"), { recursive: true });
      fs.appendFileSync(path.join(scratch, ".git", "info", "exclude"), "/node_modules\n");
    }

    deps.before?.(scratch);

    const rows: PlayRow[] = [];

    for (const step of input.steps) {
      const card = byId.get(step.card);
      if (card === undefined) {
        continue;
      }

      for (const [target, kind] of Object.entries(step.put)) {
        const dir = kind === "stub" ? input.stubDir : input.refDir;
        const dest = path.join(scratch, target);
        fs.mkdirSync(path.dirname(dest), { recursive: true });
        fs.copyFileSync(path.join(dir, target), dest);
      }

      const acceptance = card.acceptance ?? "";
      const start = deps.now();
      const result = await runAcceptance(
        acceptance,
        scratch,
        deps.timeoutMs === undefined
          ? { env: deps.env }
          : { env: deps.env, timeoutMs: deps.timeoutMs },
      );
      const seconds = Math.round((deps.now() - start) / 100) / 10;

      const isStub = step.phase === "stub";
      const verdict = isStub
        ? stubVerdict(result.log, expectedStage(acceptance), card.targets, fileLines)
        : stubVerdict(result.log, "", card.targets, fileLines, null);
      const ok = isStub
        ? result.exit !== 0 && verdict.stage === verdict.expected && verdict.outside.length === 0
        : result.exit === 0;

      rows.push({
        phase: step.phase,
        generation: step.generation,
        card: step.card,
        exit: result.exit,
        timedOut: result.timedOut,
        seconds,
        stage: verdict.stage,
        expected: isStub ? verdict.expected : null,
        outside: verdict.outside,
        failures: verdict.failures,
        ok,
      });

      if (step.commit.length > 0) {
        gitOk(scratch, ["add", "-A", "-f", "--", ...step.commit], deps.env);
        gitOk(
          scratch,
          [
            "-c",
            "user.name=morph gate",
            "-c",
            "user.email=gate@morph.invalid",
            "commit",
            "-q",
            "--allow-empty",
            "-m",
            "gate: " + step.card,
          ],
          deps.env,
        );
      }
    }

    return rows;
  } finally {
    fs.rmSync(base, { recursive: true, force: true });
  }
}
