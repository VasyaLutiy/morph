export interface StubVerdict {
  stage: string | null;
  expected: string;
  outside: string[];
  failures: string[];
}

export function expectedStage(acceptance: string): string {
  for (const line of acceptance.split("\n")) {
    if (line.startsWith("echo '== probe'")) {
      return "probe";
    }
  }
  return "guard";
}

export function stubVerdict(
  log: string,
  expected: string,
  targets: readonly string[],
  fileLines: readonly string[],
  stages: readonly string[] | null = ["build", "vet", "tsc"],
): StubVerdict {
  const outside: string[] = [];
  const failures: string[] = [];
  const seenOutside = new Set<string>();
  const seenFailures = new Set<string>();
  const targetSet = new Set<string>(targets);
  const patterns = fileLines.map((source) => new RegExp(source));
  let current: string | null = null;

  for (const raw of log.split("\n")) {
    const line = raw.replace(/\s+$/, "");
    const header = /^== (\S+)/.exec(line);
    if (header !== null) {
      current = line.includes(" failed (see above)") ? "verdict" : header[1] ?? "";
      continue;
    }
    if (current === null) {
      continue;
    }

    const t = line.trim();
    if (current === expected && (t.startsWith("FAIL ") || t.startsWith("--- FAIL: "))) {
      if (!seenFailures.has(t)) {
        seenFailures.add(t);
        failures.push(t);
      }
    }

    if (stages !== null && !stages.includes(current)) {
      continue;
    }
    if (line === "" || line.startsWith(" ") || line.startsWith("\t")) {
      continue;
    }

    for (const pattern of patterns) {
      const match = pattern.exec(line);
      if (match === null) {
        continue;
      }
      const file = match[1] ?? "";
      if (!targetSet.has(file) && !seenOutside.has(line)) {
        seenOutside.add(line);
        outside.push(line);
      }
      break;
    }
  }

  return { stage: current, expected, outside, failures };
}
