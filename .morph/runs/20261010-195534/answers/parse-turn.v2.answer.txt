/**
 * Parse Turn — one model turn of the scout protocol read as one action (the
 * lines not run named), one final answer (with or without the verb) or a
 * malformed turn with its reason. Pure: imports nothing, touches no clock, no
 * environment, no file system.
 */

export type ScoutAction =
  | { kind: "read"; path: string; from: number | null; to: number | null }
  | { kind: "grep"; pattern: string; path: string }
  | { kind: "list"; path: string };

export interface ScoutAnswer {
  targets: string[];
  context_slice: string[];
  reasoning: string;
}

export type Turn =
  | { kind: "action"; action: ScoutAction; skipped?: string[] }
  | { kind: "answer"; answer: ScoutAnswer }
  | { kind: "malformed"; reason: string };

export const VERBS: readonly string[] = ["READ", "GREP", "LIST", "ANSWER"];

const ACTION_LINE = /^(READ|GREP|LIST|ANSWER)(?=\s|$)/;
const READ_RANGE = /^(.*\S)\s+(\d+)-(\d+)$/;
const GREP_LITERAL = /^\/(.+)\/[a-z]*$/;
const NO_ACTION =
  "no action: one line must start with READ, GREP, LIST or ANSWER";

function malformed(reason: string): Turn {
  return { kind: "malformed", reason };
}

function isNonEmptyString(value: unknown): value is string {
  return typeof value === "string" && value.length > 0;
}

function isPathList(value: unknown): value is string[] {
  return Array.isArray(value) && value.every(isNonEmptyString);
}

/**
 * From the "{" at `start`, the text up to the "}" that closes it, counting
 * braces only outside JSON strings. A string runs from a '"' to the next '"'
 * not escaped by a backslash. Null when no "}" ever closes it.
 */
function balancedObject(text: string, start: number): string | null {
  let depth = 0;
  let inString = false;
  let escaped = false;
  for (let i = start; i < text.length; i++) {
    const c = text[i];
    if (inString) {
      if (escaped) {
        escaped = false;
      } else if (c === "\\") {
        escaped = true;
      } else if (c === '"') {
        inString = false;
      }
      continue;
    }
    if (c === '"') {
      inString = true;
    } else if (c === "{") {
      depth += 1;
    } else if (c === "}") {
      depth -= 1;
      if (depth === 0) {
        return text.slice(start, i + 1);
      }
    }
  }
  return null;
}

type JsonRead =
  | { status: "ok"; value: unknown }
  | { status: "noobject" }
  | { status: "unparsed" };

/**
 * The JSON object inside `text`: the first balanced object when one closes and
 * parses, otherwise the text from the first "{" to the last "}".
 */
function readJson(text: string): JsonRead {
  const a = text.indexOf("{");
  if (a < 0) {
    return { status: "noobject" };
  }
  const b = text.lastIndexOf("}");
  if (b < a) {
    return { status: "noobject" };
  }
  const balanced = balancedObject(text, a);
  if (balanced !== null) {
    try {
      return { status: "ok", value: JSON.parse(balanced) as unknown };
    } catch {
      // fall through to the loose slice
    }
  }
  try {
    return { status: "ok", value: JSON.parse(text.slice(a, b + 1)) as unknown };
  } catch {
    return { status: "unparsed" };
  }
}

function validateAnswer(value: unknown): Turn {
  if (value === null || typeof value !== "object" || Array.isArray(value)) {
    return malformed("ANSWER: targets must be a non-empty list of paths");
  }
  const obj = value as Record<string, unknown>;

  const rawTargets = obj["targets"];
  if (!isPathList(rawTargets) || rawTargets.length < 1) {
    return malformed("ANSWER: targets must be a non-empty list of paths");
  }

  const rawSlice = obj["context_slice"];
  let context_slice: string[];
  if (rawSlice === undefined) {
    context_slice = [];
  } else if (isPathList(rawSlice)) {
    context_slice = rawSlice;
  } else {
    return malformed("ANSWER: context_slice must be a list of paths");
  }

  const rawReasoning = obj["reasoning"];
  let reasoning: string;
  if (rawReasoning === undefined) {
    reasoning = "";
  } else if (typeof rawReasoning === "string") {
    reasoning = rawReasoning;
  } else {
    return malformed("ANSWER: reasoning must be a string");
  }

  return {
    kind: "answer",
    answer: { targets: rawTargets, context_slice, reasoning },
  };
}

function parseAnswer(rest: string): Turn {
  const read = readJson(rest);
  if (read.status === "noobject") {
    return malformed("ANSWER: no JSON object");
  }
  if (read.status === "unparsed") {
    return malformed("ANSWER: the JSON does not parse");
  }
  return validateAnswer(read.value);
}

function parseBareAnswer(text: string): Turn {
  const read = readJson(text);
  if (read.status !== "ok") {
    return malformed(NO_ACTION);
  }
  const value = read.value;
  if (value === null || typeof value !== "object" || Array.isArray(value)) {
    return malformed(NO_ACTION);
  }
  const obj = value as Record<string, unknown>;
  if (!Object.prototype.hasOwnProperty.call(obj, "targets")) {
    return malformed(NO_ACTION);
  }
  return validateAnswer(value);
}

function parseActionLine(verb: string, line: string): Turn {
  const args = line.slice(verb.length).trim();

  if (verb === "READ") {
    if (args === "") {
      return malformed("READ needs a path");
    }
    const range = READ_RANGE.exec(args);
    if (range !== null) {
      const path = range[1];
      const from = Number(range[2]);
      const to = Number(range[3]);
      if (from < 1 || to < from) {
        return malformed(`READ: bad line range ${from}-${to}`);
      }
      return { kind: "action", action: { kind: "read", path, from, to } };
    }
    return {
      kind: "action",
      action: { kind: "read", path: args, from: null, to: null },
    };
  }

  if (verb === "GREP") {
    const cut = args.lastIndexOf(" -- ");
    let pattern: string;
    let path: string;
    if (cut >= 0) {
      pattern = args.slice(0, cut).trim();
      path = args.slice(cut + 4).trim();
    } else {
      pattern = args;
      path = "";
    }
    const literal = GREP_LITERAL.exec(pattern);
    if (literal !== null) {
      pattern = literal[1];
    }
    if (pattern === "") {
      return malformed("GREP needs a pattern");
    }
    try {
      new RegExp(pattern);
    } catch {
      return malformed(`GREP: invalid pattern /${pattern}/`);
    }
    return { kind: "action", action: { kind: "grep", pattern, path } };
  }

  return { kind: "action", action: { kind: "list", path: args } };
}

export function parseTurn(text: string | null): Turn {
  if (text === null || text.trim() === "") {
    return malformed("empty turn");
  }

  const lines = text
    .split("\n")
    .map((line) => (line.endsWith("\r") ? line.slice(0, -1) : line));

  const found: { verb: string; index: number; line: string }[] = [];
  lines.forEach((raw, index) => {
    const match = ACTION_LINE.exec(raw.trimStart());
    if (match !== null) {
      found.push({ verb: match[1], index, line: raw });
    }
  });

  const answerEntry = found.find((entry) => entry.verb === "ANSWER");
  if (answerEntry !== undefined) {
    const rest = [
      answerEntry.line.trimStart().slice(answerEntry.verb.length),
      ...lines.slice(answerEntry.index + 1),
    ].join("\n");
    return parseAnswer(rest);
  }

  if (found.length === 0) {
    return parseBareAnswer(text);
  }

  const unique: { verb: string; line: string }[] = [];
  const seen = new Set<string>();
  for (const entry of found) {
    const trimmed = entry.line.trim();
    if (seen.has(trimmed)) {
      continue;
    }
    seen.add(trimmed);
    unique.push({ verb: entry.verb, line: trimmed });
  }

  const actionTurn = parseActionLine(unique[0].verb, unique[0].line);
  if (actionTurn.kind === "malformed") {
    return actionTurn;
  }

  const skipped = unique.slice(1).map((entry) => entry.line);
  if (skipped.length === 0) {
    return actionTurn;
  }
  return { kind: "action", action: actionTurn.action, skipped };
}
