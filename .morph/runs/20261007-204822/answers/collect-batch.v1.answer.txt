import { readFileSync, statSync } from "node:fs";
import { join } from "node:path";

import { captureInputs, compareCaptures } from "../compiler/capture.js";
import { assembleBatch } from "../processor/batchRequest.js";
import { readBatch } from "../processor/batchResponse.js";
import { realTransport } from "../processor/send.js";
import { batchConfig, refusal } from "./submit.js";
import type { DetachedDeps, DetachedResult, SubmittedBatch } from "./submit.js";

export interface CollectedAnswer {
  customId: string;
  finishReason: string | null;
  error: string | null;
  chars: number | null;
}

export interface CollectDocument {
  batchId: string;
  outcome: "pending" | "done" | "failed";
  status: string;
  cost: number | null;
  answers: CollectedAnswer[];
  stale: string[];
  files: string[];
  state: string;
}

function isPlainObject(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

function errorMessage(error: unknown): string {
  return error instanceof Error ? error.message : String(error);
}

export async function collectBatch(
  root: string,
  batchId: string,
  deps: DetachedDeps
): Promise<DetachedResult> {
  const rel = ".morph/batches/" + batchId + ".json";
  const full = join(root, rel);

  let text = "";
  try {
    if (!statSync(full).isFile()) {
      return refusal(4, "UsageError", "batch state not found: " + rel);
    }
    text = readFileSync(full, "utf8");
  } catch {
    return refusal(4, "UsageError", "batch state not found: " + rel);
  }

  const invalid = (name: string): DetachedResult =>
    refusal(2, "RefusalError", "batch state " + rel + " is not a submit's state: " + name);

  let parsed: unknown = undefined;
  try {
    parsed = JSON.parse(text);
  } catch {
    parsed = undefined;
  }
  if (!isPlainObject(parsed)) return invalid("unreadable");
  const raw = parsed;
  if (raw.batchId !== batchId) return invalid("batchId");
  if (typeof raw.processor !== "string") return invalid("processor");
  if (!Array.isArray(raw.customIds)) return invalid("customIds");
  for (const customId of raw.customIds) {
    if (typeof customId !== "string") return invalid("customIds");
  }
  if (!Array.isArray(raw.cards)) return invalid("cards");
  if (!isPlainObject(raw.inputs)) return invalid("inputs");

  const state = raw as unknown as SubmittedBatch;

  const config = batchConfig(deps.env, state.processor);
  if ("code" in config) return config;

  const call = assembleBatch([], config);
  const transport = deps.transport ?? realTransport(config.timeoutMs);

  let status = 0;
  let body = "";
  try {
    const reply = await transport.fetch(call.url + "/" + batchId, {
      method: "GET",
      headers: call.headers,
    });
    status = reply.status;
    body = await reply.text();
  } catch (error) {
    return refusal(3, "RuntimeError", "batch " + batchId + ": transport: " + errorMessage(error));
  }

  const read = readBatch(status, body);
  if (read.state === "error") {
    return refusal(3, "RuntimeError", "batch " + batchId + ": " + (read.error ?? "unknown error"));
  }

  if (read.status !== null) state.status = read.status;
  if (read.cost !== undefined) state.cost = read.cost;

  if (read.state === "pending") {
    deps.saveState(state);
    const document: CollectDocument = {
      batchId,
      outcome: "pending",
      status: state.status,
      cost: state.cost,
      answers: [],
      stale: [],
      files: [],
      state: rel,
    };
    return { code: 1, document };
  }

  const outcome: "done" | "failed" = read.state === "failed" ? "failed" : "done";

  const stale: string[] = [];
  for (const card of state.cards) {
    const before = state.inputs[card.customId] ?? {};
    if (compareCaptures(before, captureInputs(card, root)).length > 0) {
      stale.push(card.customId);
    }
  }
  const staleIds = new Set(stale);

  const answers: CollectedAnswer[] = [];
  const kept: { customId: string; text: string }[] = [];
  for (const customId of state.customIds) {
    const reply = read.replies.find((candidate) => candidate.answer.customId === customId);
    if (reply === undefined) {
      answers.push({
        customId,
        finishReason: null,
        error: "batch " + batchId + " " + state.status + ": " + (read.error ?? "no result"),
        chars: null,
      });
      continue;
    }
    const cardId = customId.replace(/\.v[0-9]+$/, "");
    answers.push({
      customId,
      finishReason: reply.answer.finishReason,
      error: reply.answer.error,
      chars: reply.answer.text === null ? null : reply.answer.text.length,
    });
    if (reply.answer.text !== null && !staleIds.has(cardId)) {
      kept.push({ customId, text: reply.answer.text });
    }
  }

  const files = deps.saveAnswers(batchId, kept);
  state.collectedAt = deps.now();
  state.files = files;
  deps.saveState(state);

  const clean =
    outcome === "done" &&
    stale.length === 0 &&
    answers.every((answer) => answer.error === null && answer.chars !== null);

  const document: CollectDocument = {
    batchId,
    outcome,
    status: state.status,
    cost: state.cost,
    answers,
    stale,
    files,
    state: rel,
  };
  return { code: clean ? 0 : 1, document };
}
