export interface StubVerdict {
  stage: string | null;
  expected: string;
  outside: string[];
  failures: string[];
}

export function expectedStage(acceptance: string): string {
  for (const line of acceptance.split("\n")) {
    if (line.startsWith("echo '== probe'")) {
      return "probe";
    }
  }
  return "guard";
}

export function stubVerdict(
  log: string,
  expected: string,
  targets: readonly string[],
  fileLines: readonly string[],
  stages: readonly string[] | null = ["build", "vet", "tsc"],
): StubVerdict {
  let current: string | null = null;
  const outside: string[] = [];
  const failures: string[] = [];
  const patterns = fileLines.map((source) => new RegExp(source));

  for (const raw of log.split("\n")) {
    const line = raw.trimEnd();

    const header = /^== (\S+)/.exec(line);
    if (header !== null) {
      current = line.includes(" failed (see above)") ? "verdict" : (header[1] ?? "");
      continue;
    }

    const text = line.trim();
    if (current === expected && (text.startsWith("FAIL ") || text.startsWith("--- FAIL: "))) {
      if (!failures.includes(text)) {
        failures.push(text);
      }
    }

    if (stages !== null && (current === null || !stages.includes(current))) {
      continue;
    }
    if (line === "" || line.startsWith(" ") || line.startsWith("\t")) {
      continue;
    }

    for (const pattern of patterns) {
      const match = pattern.exec(line);
      if (match === null) {
        continue;
      }
      const file = match[1] ?? "";
      if (!targets.includes(file) && !outside.includes(line)) {
        outside.push(line);
      }
      break;
    }
  }

  return { stage: current, expected, outside, failures };
}
