import * as fs from "node:fs";
import * as os from "node:os";
import * as path from "node:path";
import { runAcceptance } from "../acceptance/run.js";
import { gitOk } from "../git/run.js";
import { TREE_PROFILES } from "../language/treeProfiles.js";
import { expectedStage, stubVerdict } from "./stubVerdict.js";
import type { Card } from "../cards/types.js";
import type { GatePhase, GateStep } from "./gatePlan.js";

export interface PlayRow {
  phase: GatePhase;
  generation: number;
  card: string;
  exit: number | null;
  timedOut: boolean;
  seconds: number;
  stage: string | null;
  expected: string | null;
  outside: string[];
  failures: string[];
  ok: boolean;
}

export interface PlayInput {
  root: string;
  cards: readonly Card[];
  steps: readonly GateStep[];
  stubDir: string;
  refDir: string;
}

export interface PlayDeps {
  env: Record<string, string>;
  now: () => number;
  timeoutMs?: number;
  before?: (scratch: string) => void;
}

const FILE_LINES: readonly string[] = TREE_PROFILES.map((profile) => profile.fileLine);

/**
 * Plays a gate in a scratch clone of root's committed HEAD: one acceptance run
 * per step, in the plan's order. Never a worktree — `git clone --shared
 * --no-checkout` borrows the objects and writes nothing into root — and the
 * scratch's base directory is removed in a finally. Every time comes from
 * deps.now, every child process from Run Acceptance and Run Git.
 */
export async function playGate(input: PlayInput, deps: PlayDeps): Promise<PlayRow[]> {
  const base = fs.mkdtempSync(path.join(os.tmpdir(), "morph-gate-"));
  const scratch = path.join(base, "w");
  try {
    gitOk(input.root, ["clone", "-q", "--shared", "--no-checkout", input.root, scratch], deps.env);
    const head = gitOk(input.root, ["rev-parse", "HEAD"], deps.env).trim();
    gitOk(scratch, ["checkout", "-q", "--detach", head], deps.env);

    const modules = path.join(input.root, "node_modules");
    if (fs.existsSync(modules)) {
      fs.symlinkSync(fs.realpathSync(modules), path.join(scratch, "node_modules"));
      const info = path.join(scratch, ".git", "info");
      fs.mkdirSync(info, { recursive: true });
      fs.appendFileSync(path.join(info, "exclude"), "/node_modules\n");
    }

    deps.before?.(scratch);

    const byId = new Map<string, Card>();
    for (const card of input.cards) {
      byId.set(card.customId, card);
    }

    const rows: PlayRow[] = [];
    for (const step of input.steps) {
      const card = byId.get(step.card);
      if (card === undefined) {
        continue;
      }

      for (const target of Object.keys(step.put)) {
        const from = path.join(step.put[target] === "stub" ? input.stubDir : input.refDir, target);
        const to = path.join(scratch, target);
        fs.mkdirSync(path.dirname(to), { recursive: true });
        fs.copyFileSync(from, to);
      }

      const acceptance = card.acceptance ?? "";
      const start = deps.now();
      const result = await runAcceptance(
        acceptance,
        scratch,
        deps.timeoutMs === undefined
          ? { env: deps.env }
          : { env: deps.env, timeoutMs: deps.timeoutMs },
      );
      const seconds = Math.round((deps.now() - start) / 100) / 10;

      const isStub = step.phase === "stub";
      const verdict = isStub
        ? stubVerdict(result.log, expectedStage(acceptance), card.targets, FILE_LINES)
        : stubVerdict(result.log, "", card.targets, FILE_LINES, null);
      const ok = isStub
        ? result.exit !== 0 && verdict.stage === verdict.expected && verdict.outside.length === 0
        : result.exit === 0;

      rows.push({
        phase: step.phase,
        generation: step.generation,
        card: step.card,
        exit: result.exit,
        timedOut: result.timedOut,
        seconds,
        stage: verdict.stage,
        expected: isStub ? verdict.expected : null,
        outside: verdict.outside,
        failures: verdict.failures,
        ok,
      });

      if (step.commit.length > 0) {
        gitOk(scratch, ["add", "-A", "-f", "--", ...step.commit], deps.env);
        gitOk(
          scratch,
          [
            "-c",
            "user.name=morph gate",
            "-c",
            "user.email=gate@morph.invalid",
            "commit",
            "-q",
            "--allow-empty",
            "-m",
            "gate: " + step.card,
          ],
          deps.env,
        );
      }
    }

    return rows;
  } finally {
    fs.rmSync(base, { recursive: true, force: true });
  }
}
