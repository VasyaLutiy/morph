import { expect, test } from "vitest";
import { goCodeAcceptance, goJudgeAcceptance } from "../../src/builder/goAcceptance.js";
import { DEFAULT_FROZEN } from "../../src/builder/readChecks.js";
import { GO } from "../../src/language/profiles.js";
import { fixture } from "../helpers.js";
import type { CardContext, JudgeFile } from "../../src/builder/types.js";

const goCtx = (over: Partial<CardContext>): CardContext => ({
  id: "percent-of",
  phase: "m1",
  targets: ["calc/percent_of.go"],
  siblings: ["calc/clamp_value_examples_test.go"],
  frozen: ["go.mod", "internal"],
  fullExclude: [],
  ownGit: false,
  profile: GO,
  guard: "// guard\n",
  firstdiff: "// firstdiff\n",
  ...over,
});

test("Go Acceptance example 8: the layer argument reaches the code guard lines, the judge keeps its own", () => {
  const files: JudgeFile[] = [
    {
      file: "calc/percent_of_examples_test.go",
      min: 5,
      max: 11,
      lits: ["TestPercentOfExample1", "0% (0 of 0)"],
      drop: [],
      new: true,
    },
  ];

  const empty = goCodeAcceptance(
    goCtx({ layers: { calc: [] } }),
    "package calc\n",
    null,
    null,
  );
  expect(empty).toBe(
    fixture("builder/go/code1.txt").replace(
      "echo '== guard'; node $P/guard.mjs src calc/percent_of.go\n",
      "echo '== guard'; node $P/guard.mjs src calc/percent_of.go '' '{\"calc\":[]}'\n",
    ),
  );

  const allowed = goCodeAcceptance(
    goCtx({
      allowed: ["github.com/dustin/go-humanize"],
      layers: { calc: ["report"] },
    }),
    "package calc\n",
    null,
    null,
  );
  expect(allowed).toBe(
    fixture("builder/go/code1.txt").replace(
      "echo '== guard'; node $P/guard.mjs src calc/percent_of.go\n",
      "echo '== guard'; node $P/guard.mjs src calc/percent_of.go 'github.com/dustin/go-humanize' '{\"calc\":[\"report\"]}'\n",
    ),
  );

  const own = goCodeAcceptance(
    goCtx({
      id: "a",
      phase: "p2",
      targets: ["report/a.go", "report/a_test.go"],
      siblings: [],
      frozen: DEFAULT_FROZEN,
      fullExclude: ["calc/old_test.go"],
      ownGit: true,
      layers: { report: ["calc"] },
    }),
    "package report\n",
    5,
    "echo '== bin'; true\n",
  );
  expect(own).toBe(
    fixture("builder/go/code2.txt").replace(
      "guard.mjs src report/a.go; node",
      "guard.mjs src report/a.go '' '{\"report\":[\"calc\"]}'; node",
    ),
  );

  const judge = goJudgeAcceptance(
    goCtx({
      id: "percent-of-judge",
      targets: ["calc/percent_of_examples_test.go"],
      siblings: ["report/format_share.go"],
      layers: { calc: [] },
    }),
    files,
  );
  expect(judge).toBe(fixture("builder/go/judge1.txt"));
});
