import fs from "node:fs";
import path from "node:path";
import { layerGenerations } from "../../src/cards/layer.js";
import type { Deck } from "../../src/cards/types.js";
import { checkBuilds } from "../../src/cli/checkBuilds.js";
import type { BuildCheck } from "../../src/cli/checkBuilds.js";
import { readDeckFile } from "../../src/cli/document.js";
import { fixture, fixtureJson, fixturePath, tmpRoot } from "../helpers.js";
import type { TmpRoot } from "../helpers.js";

function copyOf(name: string): TmpRoot {
  const r = tmpRoot();
  fs.cpSync(fixturePath(name), r.root, { recursive: true });
  return r;
}

function rootOf(name: string): TmpRoot {
  const r = tmpRoot();
  for (const [p, t] of Object.entries(fixtureJson(name) as Record<string, string>)) r.write(p, t);
  fs.symlinkSync(path.resolve(fixturePath("."), "../../node_modules"), r.path("node_modules"));
  return r;
}

function load(r: TmpRoot, deckPath: string): { deck: Deck; generations: string[][] } {
  const l = readDeckFile(r.root, deckPath);
  if (!l.ok) throw new Error(deckPath);
  return { deck: l.deck, generations: layerGenerations(l.deck) };
}

const pathEnv = (): Record<string, string> => ({ PATH: process.env.PATH ?? "" });

function scriptsRoot(): { r: TmpRoot; env: Record<string, string> } {
  const r = tmpRoot();
  r.write("d.json", fixture("planner/stub.cards.json"));
  for (const f of ["calc/a.go", "report/c.go", "src/t.ts", "tests/t.test.ts"]) r.write("_stubs/" + f, "x\n");
  r.write("bin/go", '#!/bin/sh\necho "vet: ./calc/zz.go:1:2: go ran with $GOFLAGS"\nexit 1\n');
  r.write(
    "node_modules/.bin/tsc",
    '#!/bin/sh\necho "src/t.ts(1,1): error TS1: own"\necho "src/zz.ts(2,3): error TS2: tsc ran with $NO_COLOR $3"\nexit 2\n',
  );
  fs.chmodSync(r.path("bin/go"), 0o755);
  fs.chmodSync(r.path("node_modules/.bin/tsc"), 0o755);
  return { r, env: { PATH: r.path("bin") + ":" + (process.env.PATH ?? "") } };
}

const builds = (k: string): BuildCheck[] => (fixtureJson("cli/checkBuilds.json") as Record<string, BuildCheck[]>)[k];

// A tmp root whose PATH finds a fake `go` that prints nothing a break
// matcher would keep (a "#" line) and exits 0.
function fakeGoRoot(): { r: TmpRoot; env: Record<string, string> } {
  const r = tmpRoot();
  r.write("bin/go", '#!/bin/sh\necho "# ok"\nexit 0\n');
  fs.chmodSync(r.path("bin/go"), 0o755);
  return { r, env: { PATH: r.path("bin") + ":" + (process.env.PATH ?? "") } };
}

test("Check Builds example 1: the P20 cut breaks, the P21 cut does not", () => {
  const r = copyOf("go-p7b");
  try {
    const a = load(r, "decks/b1/deck.p20.json");
    expect(
      checkBuilds(r.root, "decks/b1/deck.p20.json", a.deck, a.generations, pathEnv()),
    ).toStrictEqual(builds("p20"));

    const b = load(r, "decks/b1/deck.p21.json");
    expect(
      checkBuilds(r.root, "decks/b1/deck.p21.json", b.deck, b.generations, pathEnv()),
    ).toStrictEqual(builds("p21"));

    fs.rmSync(r.path("decks/b1/_stubs/supervisor/guard.go"));
    expect(
      checkBuilds(r.root, "decks/b1/deck.p21.json", b.deck, b.generations, pathEnv()),
    ).toStrictEqual(builds("p21 without the guard.go stub"));
  } finally {
    r.rm();
  }
}, 120000);

test("checkBuilds is null when the deck's directory has no _stubs", () => {
  const r = tmpRoot();
  try {
    r.write("d.json", DECK_GO);
    const l = load(r, "d.json");
    expect(checkBuilds(r.root, "d.json", l.deck, l.generations, pathEnv())).toBe(null);

    r.rm();
  }
});

test("checkBuilds reads _stubs beside the deck file's own directory", () => {
  const s = fakeGoRoot();
  try {
    s.r.write("sub/d.json", DECK_GO);
    s.r.write("sub/_stubs/calc/a.go", "x\n");
    expect(s.r.exists("_stubs")).toBe(false);
    const l = load(s.r, "sub/d.json");
    const got = checkBuilds(s.r.root, "sub/d.json", l.deck, l.generations, s.env);
    expect(got).toStrictEqual([
      {
        card: "one",
        generation: 0,
        language: "go",
        stubbed: 1,
        missing: [],
        breaks: [],
        note: null,
      },
    ]);
  } finally {
    s.r.rm();
  }
}, 120000);

test("checkBuilds does not count a symbolic link in _stubs as a stub", () => {
  const s = fakeGoRoot();
  try {
    s.r.write("d.json", DECK_GO);
    s.r.write("_stubs/real.go", "x\n");
    fs.mkdirSync(s.r.path("_stubs/calc"), { recursive: true });
    fs.symlinkSync("../real.go", s.r.path("_stubs/calc/a.go"));
    const l = load(s.r, "d.json");
    const got = checkBuilds(s.r.root, "d.json", l.deck, l.generations, s.env);
    expect(got).toStrictEqual([
      {
        card: "one",
        generation: 0,
        language: "go",
        stubbed: 0,
        missing: ["calc/a.go"],
        breaks: [],
        note: null,
      },
    ]);
  } finally {
    s.r.rm();
  }
}, 120000);
