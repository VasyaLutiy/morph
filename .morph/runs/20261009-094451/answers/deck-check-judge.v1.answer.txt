import fs from "node:fs";
import path from "node:path";
import { expect, test } from "vitest";
import { deckCheckCommand } from "../../src/cli/deckCheck.js";
import { main } from "../../src/cli/main.js";
import { fixture, fixtureJson, fixturePath, tmpRoot } from "../helpers.js";
import type { TmpRoot } from "../helpers.js";
import type { BuildCheck } from "../../src/cli/checkBuilds.js";
import type { CliDeps, CliIo, DeckCheckDocument } from "../../src/cli/types.js";

// A copy of a fixture directory, the whole subtree.
function copyOf(name: string): TmpRoot {
  const r = tmpRoot();
  fs.cpSync(fixturePath(name), r.root, { recursive: true });
  return r;
}

// A root built from a fixture that is a map path -> text, with node_modules.
function rootOf(name: string): TmpRoot {
  const r = tmpRoot();
  for (const [p, t] of Object.entries(fixtureJson(name) as Record<string, string>)) r.write(p, t);
  fs.symlinkSync(path.resolve(fixturePath("."), "../../node_modules"), r.path("node_modules"));
  return r;
}

const pathEnv = (): Record<string, string> => ({ PATH: process.env.PATH ?? "" });

const builds = (key: string): BuildCheck[] =>
  (fixtureJson("cli/checkBuilds.json") as Record<string, BuildCheck[]>)[key];

// Deck Check 2's document: one TypeScript card with no config written.
const DC2 = {
  deck: "d.json",
  cards: 1,
  generations: [["a"]],
  errors: 0,
  warnings: 1,
  hazards: [
    { kind: "implicit-read", severity: "warning", cards: ["a"], path: null, repair: null },
  ],
  weights: [{ card: "a", bytes: 20, missing: [] }],
};

// A deck of scripted cards whose stubs are there and whose steps are scripts.
function scriptsRoot(): { r: TmpRoot; env: Record<string, string> } {
  const r = tmpRoot();
  r.write("d.json", fixture("planner/stub.cards.json"));
  for (const f of ["calc/a.go", "report/c.go", "src/t.ts", "tests/t.test.ts"]) r.write("_stubs/" + f, "x\n");
  r.write("bin/go", '#!/bin/sh\necho "vet: ./calc/zz.go:1:2: go ran with $GOFLAGS"\nexit 1\n');
  r.write(
    "node_modules/.bin/tsc",
    '#!/bin/sh\necho "src/t.ts(1,1): error TS1: own"\necho "src/zz.ts(2,3): error TS2: tsc ran with $NO_COLOR $3"\nexit 2\n',
  );
  fs.chmodSync(r.path("bin/go"), 0o755);
  fs.chmodSync(r.path("node_modules/.bin/tsc"), 0o755);
  return { r, env: { PATH: r.path("bin") + ":" + (process.env.PATH ?? "") } };
}

test(
  "Deck Check example 4: the old cut's breaks, the new cut clean",
  () => {
    const r = copyOf("go-p7b");
    const env = pathEnv();
    try {
      const old = deckCheckCommand(r.root, "decks/b1/deck.p20.json", 500000, env);
      expect(old.code).toBe(2);
      const oldDoc = old.document as DeckCheckDocument;
      expect(oldDoc.errors).toBe(9);
      expect(oldDoc.warnings).toBe(0);
      expect(oldDoc.builds).toStrictEqual(builds("p20"));
      expect(Object.keys(oldDoc)[Object.keys(oldDoc).length - 1]).toBe("builds");

      const fresh = deckCheckCommand(r.root, "decks/b1/deck.p21.json", 500000, env);
      expect(fresh.code).toBe(0);
      const freshDoc = fresh.document as DeckCheckDocument;
      expect(freshDoc.errors).toBe(0);
      expect(freshDoc.builds).toStrictEqual(builds("p21"));
    } finally {
      r.rm();
    }
  },
  120000,
);

test(
  "Deck Check example 5: the rename's breaks, a card with no config noted",
  () => {
    const r = rootOf("ts-rename.json");
    const env = pathEnv();
    try {
      const got = deckCheckCommand(r.root, "decks/r1/deck.json", 500000, env);
      expect(got.code).toBe(2);
      const doc = got.document as DeckCheckDocument;
      expect(doc.errors).toBe(5);
      expect(doc.builds).toStrictEqual(builds("ts-rename"));
    } finally {
      r.rm();
    }

    const t = tmpRoot();
    try {
      t.write("d.json", fixture("decks/tiny.json"));
      t.write("src/a.ts", "export const a = 1;\n");
      t.write("_stubs/src/a.ts", "x\n");
      const got = deckCheckCommand(t.root, "d.json", 500000, env);
      expect(got.code).toBe(0);
      const doc = got.document as DeckCheckDocument;
      expect(doc).toStrictEqual({
        ...DC2,
        warnings: 2,
        builds: [
          {
            card: "a",
            generation: 0,
            language: "typescript",
            stubbed: 0,
            missing: [],
            breaks: [],
            note: "the acceptance writes no $P/tsconfig.card.json: not built",
          },
        ],
      });
    } finally {
      t.rm();
    }
  },
  120000,
);

test(
  "Deck Check example 6: main, scripted steps, notes counted as warnings",
  async () => {
    const { r, env } = scriptsRoot();
    const out: string[] = [];
    const err: string[] = [];
    const io: CliIo = {
      stdout: (t) => {
        out.push(t);
      },
      stderr: (t) => {
        err.push(t);
      },
    };
    const deps: CliDeps = { env, now: () => 0, cwd: "/", transport: null };
    try {
      const code = await main(["deck", "check", "--deck", "d.json", "--root", r.root], deps, io);
      expect(code).toBe(2);
      expect(out.length).toBe(1);
      const doc = JSON.parse(out.join("")) as DeckCheckDocument;
      expect(doc.errors).toBe(17);
      expect(doc.warnings).toBe(12);
      expect(doc.builds).toStrictEqual(builds("cards"));
      expect(err).toStrictEqual(["morph deck check: exit 2\n"]);
    } finally {
      r.rm();
    }
  },
  120000,
);
