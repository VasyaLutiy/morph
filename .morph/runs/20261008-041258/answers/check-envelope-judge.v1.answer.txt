```ts
import { test, expect } from "vitest";
import { ENVELOPE_SOURCE, checkEnvelope, readDiff } from "../../src/reviewer/checkEnvelope.js";
import type {
  ChangedFile,
  RangeCommit,
  ScoutScope,
} from "../../src/reviewer/checkEnvelope.js";
import { readOwnership } from "../../src/git/ownership.js";

const HISTORY: RangeCommit[] = [
  { card: "a-judge", model: "m/x", run: "r2", paths: ["tests/a.examples.test.ts"] },
  { card: "a.r1", model: "m/x", run: "r2", paths: ["src/a.ts"] },
  { card: "b", model: "", run: null, paths: ["src/b.ts", "src/a.ts"] },
];

const CHANGED: ChangedFile[] = [
  { path: "src/a.ts", status: "modified", added: 3, deleted: 1 },
  { path: "src/b.ts", status: "modified", added: 1, deleted: 1 },
  { path: "README.md", status: "added", added: 2, deleted: 0 },
  { path: ".morph/runs/r2/report.json", status: "added", added: 40, deleted: 0 },
  { path: "tests/a.examples.test.ts", status: "added", added: 30, deleted: 0 },
];

const SCOUT: ScoutScope = {
  scoutId: "20261008-225320-74e423b1",
  targets: ["src/a.ts", "src/c.ts"],
  contextSlice: ["src/b.ts"],
};

test("Check Envelope example 1: readDiff pairs status letters with numstat counts", () => {
  const first = readDiff(
    "M\0a.ts\0M\0bin.dat\0D\0old.ts\0A\0src/sp ace.ts\0",
    "2\t1\ta.ts\0-\t-\tbin.dat\0" + "0\t1\told.ts\0" + "1\t0\tsrc/sp ace.ts\0",
  );
  expect(first).toStrictEqual([
    { path: "a.ts", status: "modified", added: 2, deleted: 1 },
    { path: "bin.dat", status: "modified", added: null, deleted: null },
    { path: "old.ts", status: "deleted", added: 0, deleted: 1 },
    { path: "src/sp ace.ts", status: "added", added: 1, deleted: 0 },
  ]);

  const second = readDiff("T\0link\0A\0new.ts\0", "12\t0\tnew.ts\0");
  expect(second).toStrictEqual([
    { path: "link", status: "modified", added: null, deleted: null },
    { path: "new.ts", status: "added", added: 12, deleted: 0 },
  ]);

  expect(readDiff("", "")).toStrictEqual([]);
});

test("Check Envelope example 2: rows, writers and owners of a range", () => {
  const result = checkEnvelope({
    changed: CHANGED,
    commits: HISTORY.slice(0, 2),
    ownership: readOwnership(HISTORY),
    scout: null,
  });
  expect(result).toStrictEqual({
    applies: true,
    rows: [
      {
        path: "src/a.ts",
        status: "modified",
        added: 3,
        deleted: 1,
        writers: ["a.r1"],
        owner: "a.r1 (m/x, run r2)",
        scope: null,
      },
      {
        path: "src/b.ts",
        status: "modified",
        added: 1,
        deleted: 1,
        writers: [],
        owner: "b (—, run —)",
        scope: null,
      },
      {
        path: "README.md",
        status: "added",
        added: 2,
        deleted: 0,
        writers: [],
        owner: null,
        scope: null,
      },
      {
        path: ".morph/runs/r2/report.json",
        status: "added",
        added: 40,
        deleted: 0,
        writers: [],
        owner: null,
        scope: null,
      },
      {
        path: "tests/a.examples.test.ts",
        status: "added",
        added: 30,
        deleted: 0,
        writers: ["a-judge"],
        owner: "a-judge (m/x, run r2)",
        scope: null,
      },
    ],
    findings: [
      {
        kind: "envelope",
        source: ENVELOPE_SOURCE,
        path: "src/b.ts",
        expected: "written by a Morph card of the range",
        got: "changed outside every card's targets; last Morph write b (—, run —)",
      },
      {
        kind: "envelope",
        source: ENVELOPE_SOURCE,
        path: "README.md",
        expected: "written by a Morph card of the range",
        got: "changed outside every card's targets; no Morph card ever wrote it",
      },
    ],
  });
});

test("Check Envelope example 3: without commits the envelope does not apply", () => {

test("Check Envelope example 4: the scout scopes and findings", () => {
  const result = checkEnvelope({
    changed: CHANGED.slice(0, 4),
    commits: HISTORY.slice(0, 2),
    ownership: readOwnership(HISTORY),
    scout: SCOUT,
  });
  expect(result.applies).toBe(true);
  expect(result.rows.map((row) => row.scope)).toStrictEqual([
    "target",
    "context",
    "outside",
    null,
  ]);
  expect(result.findings).toStrictEqual([
    {
      kind: "envelope",
      source: ENVELOPE_SOURCE,
      path: "src/b.ts",
      expected: "written by a Morph card of the range",
      got: "changed outside every card's targets; last Morph write b (—, run —)",
    },
    {
      kind: "scope",
      source: "scout 20261008-225320-74e423b1",
      path: "src/b.ts",
      expected: "a target of the scout session (src/a.ts, src/c.ts)",
      got: "changed, though the scout named it as context",
    },
    {
      kind: "envelope",
      source: ENVELOPE_SOURCE,
      path: "README.md",
      expected: "written by a Morph card of the range",
      got: "changed outside every card's targets; no Morph card ever wrote it",
    },
    {
      kind: "scope",
      source: "scout 20261008-225320-74e423b1",
      path: "README.md",
      expected: "a target of the scout session (src/a.ts, src/c.ts)",
      got: "changed, though the scout did not name it",
    },
    {
      kind: "scope",
      source: "scout 20261008-225320-74e423b1",
      path: "src/c.ts",
      expected: "changed: the scout named it as a target",
      got: "unchanged in the range",
    },
  ]);
});
```
