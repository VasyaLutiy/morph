export interface MutationRule {
  from: string;
  to: string;
  word: boolean;
}

export interface Mutant {
  path: string;
  line: number;
  column: number;
  rule: string;
  text: string;
}

export const MUTATION_RULES: readonly MutationRule[] = [
  { from: "===", to: "!==", word: false },
  { from: "!==", to: "===", word: false },
  { from: " <= ", to: " < ", word: false },
  { from: " >= ", to: " > ", word: false },
  { from: " < ", to: " <= ", word: false },
  { from: " > ", to: " >= ", word: false },
  { from: " + ", to: " - ", word: false },
  { from: " - ", to: " + ", word: false },
  { from: "&&", to: "||", word: false },
  { from: "||", to: "&&", word: false },
  { from: "true", to: "false", word: true },
  { from: "false", to: "true", word: true },
];

const SKIP_LINE = /^\s*(?:\/\/|\/\*|\*|#|import\b|export\s+(?:type\s+)?\{)/;
const WORD_CHAR = /[A-Za-z0-9_$]/;

export function quotedMask(line: string): boolean[] {
  const mask: boolean[] = new Array<boolean>(line.length).fill(false);
  let i = 0;
  while (i < line.length) {
    const quote = line[i];
    if (quote === '"' || quote === "'" || quote === "`") {
      mask[i] = true;
      i += 1;
      while (i < line.length) {
        const ch = line[i];
        if (ch === "\\") {
          mask[i] = true;
          if (i + 1 < line.length) mask[i + 1] = true;
          i += 2;
        } else {
          mask[i] = true;
          i += 1;
          if (ch === quote) break;
        }
      }
    } else {
      i += 1;
    }
  }
  return mask;
}

interface Occurrence {
  index: number;
  ruleIndex: number;
}

export function planMutants(path: string, text: string, limit: number): Mutant[] {
  if (limit <= 0) return [];
  const lines = text.split("\n");
  const all: Mutant[] = [];
  for (let lineIndex = 0; lineIndex < lines.length; lineIndex++) {
    const line = lines[lineIndex];
    if (SKIP_LINE.test(line)) continue;
    const mask = quotedMask(line);
    let codeEnd = line.length;
    for (let i = 0; i + 1 < line.length; i++) {
      if (line[i] === "/" && line[i + 1] === "/" && mask[i] !== true && mask[i + 1] !== true) {
        codeEnd = i;
        break;
      }
    }
    const occurrences: Occurrence[] = [];
    for (let ruleIndex = 0; ruleIndex < MUTATION_RULES.length; ruleIndex++) {
      const rule = MUTATION_RULES[ruleIndex];
      let start = 0;
      for (;;) {
        const index = line.indexOf(rule.from, start);
        if (index < 0) break;
        start = index + 1;
        if (index + rule.from.length > codeEnd) continue;
        if (mask[index] === true) continue;
        if (rule.word) {
          const before = index > 0 ? line[index - 1] : "";
          const after =
            index + rule.from.length < line.length ? line[index + rule.from.length] : "";
          if (WORD_CHAR.test(before)) continue;
          if (WORD_CHAR.test(after)) continue;
        }
        occurrences.push({ index, ruleIndex });
      }
    }
    occurrences.sort((a, b) => a.index - b.index || a.ruleIndex - b.ruleIndex);
    for (const occurrence of occurrences) {
      const rule = MUTATION_RULES[occurrence.ruleIndex];
      const offset = rule.from.length - rule.from.trimStart().length;
      const mutatedLine =
        line.slice(0, occurrence.index) + rule.to + line.slice(occurrence.index + rule.from.length);
      const mutatedLines = lines.slice();
      mutatedLines[lineIndex] = mutatedLine;
      all.push({
        path,
        line: lineIndex + 1,
        column: occurrence.index + offset + 1,
        rule: rule.from.trim() + " → " + rule.to.trim(),
        text: mutatedLines.join("\n"),
      });
    }
  }
  if (all.length <= limit) return all;
  const picked: Mutant[] = [];
  for (let i = 0; i < limit; i++) {
    picked.push(all[Math.floor((i * all.length) / limit)]);
  }
  return picked;
}
