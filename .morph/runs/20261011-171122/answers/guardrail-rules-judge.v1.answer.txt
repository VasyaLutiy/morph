test("Rules Arg example 4: the old guard and the record-driven guard agree", () => {
  const trees = fixtureJson("rules/parity/trees.json") as Record<
    string,
    Record<string, string>
  >;
  const cases: [string, string[], number[]][] = [
    ["go", ["calc/add.go", "clock/now.go", "shell/run.go", "app/app.go"], [0, 1, 1, 0]],
    [
      "typescript",
      [
        "src/core/add.ts",
        "src/clock/now.ts",
        "src/net/page.ts",
        "src/app/run.ts",
        "src/util/pad.ts",
        "src/main/cli.ts",
      ],
      [0, 1, 1, 1, 0, 0],
    ],
  ];
  for (const [lang, list, want] of cases) {
    const R = loadContour(fixture("rules/parity/" + lang + "/contour.yaml"), "contour.yaml");
    const M = loadMap(fixture("rules/parity/" + lang + "/morph-map.json"), "morph-map.json");
    if (!R.ok || !M.ok) throw new Error("fixture");
    const edges = layerEdges(R.record, M.map);
    if (!edges.ok) throw new Error("edges");
    const rules = guardrailRules(R.record, M.map);
    if (!rules.ok) throw new Error("rules");
    const tree = trees[lang];
    if (tree === undefined) throw new Error("tree");
    const r = tmpRoot();
    try {
      for (const [p, text] of Object.entries(tree)) r.write(p, text);
      r.write("old-guard.mjs", fixture("rules/guard2/" + lang + ".guard.txt"));
      fs.symlinkSync(path.join(ROOT, "node_modules"), r.path("node_modules"));
      for (let i = 0; i < list.length; i++) {
        const file = list[i];
        const expected = want[i];
        const oldRun = spawnSync("node", [r.path("old-guard.mjs"), "src", file], {
          cwd: r.root,
          encoding: "utf8",
          env: ENV,
        });
        const newRun = run(r.root, lang, [
          "src",
          file,
          "",
          JSON.stringify(edges.edges),
          JSON.stringify(rules.rules),
        ]);
        expect(oldRun.status).toBe(expected);
        expect(newRun.status).toBe(expected);
      }
    } finally {
      r.rm();
    }
  }
}, 60000);

test("Layers Arg example 2: the Go guard reads the record's layers", () => {
  const r = tmpRoot();
  try {
    r.write("go.mod", "module m\n\ngo 1.22\n");
    r.write("x/x.go", "package x\n\nconst X = 1\n");
    r.write("y/y.go", "package y\n\nconst Y = 2\n");
    r.write("queue/q.go", "package queue\n\nconst Q = 3\n");
    r.write("users/u.go", "package users\n\nconst U = 4\n");
    r.write("pkg/a.go", "package pkg\n\nimport \"m/x\"\n\nconst A = x.X\n");
    r.write("pkg/b.go", "package pkg\n\nimport \"m/y\"\n\nconst B = y.Y\n");
    r.write(
      "registry/x.go",
      "package registry\n\nimport (\n\t\"m/queue\"\n\t\"m/users\"\n)\n\nconst R = queue.Q + users.U\n",
    );
    r.write(
      "decks/tools/layers.json",
      "{\"layers\":{\"registry\":[\"queue\",\"users\"]},\"pure\":[]}\n",
    );

    const L1 = { pkg: ["x", "y"] };
    const a = run(r.root, "go", ["src", "pkg/a.go", "", JSON.stringify(L1), "{}"]);
    expect(a.status).toBe(0);
    expect(a.stdout).toBe("");
    const b = run(r.root, "go", ["src", "pkg/b.go", "", JSON.stringify(L1), "{}"]);
    expect(b.status).toBe(0);
    expect(b.stdout).toBe("");

    const L2 = { registry: ["queue"] };
    const red = run(r.root, "go", ["src", "registry/x.go", "", JSON.stringify(L2), "{}"]);
    expect(red.status).toBe(1);
    expect(red.stdout).toBe(
      "guard: package registry imports m/users: the record has no 'uses' from registry to users (allowed: queue)\n",
    );
  } finally {
    r.rm();
  }
}, 60000);
