import { posix } from "node:path";

export interface CodeText {
  path: string;
  text: string;
}

export interface LayerGap {
  from: string;
  to: string;
  file: string;
}

const TS_SPECIFIER = /\bfrom\s*["']([^"']+)["']|\bimport\s*["']([^"']+)["']/g;

const GO_BLOCK_START = /^\s*import\s*\(/;
const GO_SINGLE = /^\s*import\s+(?:[\w.]+\s+)?"([^"]+)"/;
const GO_BLOCK_LINE = /^\s*(?:[\w.]+\s+)?"([^"]+)"/;

function typescriptTargets(dir: string, text: string): string[] {
  const targets: string[] = [];
  TS_SPECIFIER.lastIndex = 0;
  let match: RegExpExecArray | null = TS_SPECIFIER.exec(text);
  while (match !== null) {
    const specifier = match[1] ?? match[2];
    if (
      specifier !== undefined &&
      (specifier.startsWith("./") || specifier.startsWith("../"))
    ) {
      targets.push(posix.dirname(posix.join(dir, specifier)));
    }
    match = TS_SPECIFIER.exec(text);
  }
  return targets;
}

function goRest(module: string, specifier: string): string | null {
  if (module === "") return null;
  const prefix = module + "/";
  if (!specifier.startsWith(prefix)) return null;
  return specifier.slice(prefix.length);
}

function goTargets(module: string, text: string): string[] {
  const targets: string[] = [];
  const lines = text.split(/\r?\n/);
  let inBlock = false;
  for (const line of lines) {
    if (inBlock) {
      if (line.trim() === ")") {
        inBlock = false;
        continue;
      }
      const block = GO_BLOCK_LINE.exec(line);
      if (block !== null) {
        const specifier = block[1];
        if (specifier !== undefined) {
          const rest = goRest(module, specifier);
          if (rest !== null) targets.push(rest);
        }
      }
      continue;
    }
    if (GO_BLOCK_START.test(line)) {
      inBlock = true;
      continue;
    }
    const single = GO_SINGLE.exec(line);
    if (single !== null) {
      const specifier = single[1];
      if (specifier !== undefined) {
        const rest = goRest(module, specifier);
        if (rest !== null) targets.push(rest);
      }
    }
  }
  return targets;
}

export function importGaps(
  language: string,
  module: string,
  files: readonly CodeText[],
  edges: Record<string, string[]>,
): LayerGap[] {
  const seen = new Set<string>();
  const gaps: LayerGap[] = [];

  for (const file of files) {
    const dir = posix.dirname(file.path);
    if (!Object.hasOwn(edges, dir)) continue;

    const allowed = edges[dir] ?? [];

    let targets: string[];
    if (language === "typescript") {
      targets = typescriptTargets(dir, file.text);
    } else if (language === "go") {
      targets = goTargets(module, file.text);
    } else {
      targets = [];
    }

    for (const to of targets) {
      if (to === dir) continue;
      if (!Object.hasOwn(edges, to)) continue;
      if (allowed.includes(to)) continue;

      const key = dir + "\u0000" + to;
      if (seen.has(key)) continue;
      seen.add(key);
      gaps.push({ from: dir, to, file: file.path });
    }
  }

  gaps.sort((a, b) => {
    if (a.from < b.from) return -1;
    if (a.from > b.from) return 1;
    if (a.to < b.to) return -1;
    if (a.to > b.to) return 1;
    return 0;
  });

  return gaps;
}
