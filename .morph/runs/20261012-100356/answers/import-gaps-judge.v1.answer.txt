import { expect, test } from "vitest";

import { importGaps } from "../../src/primerRecord/importGaps.js";

const EX = fixtureJson("primerRecord/examples.json") as Record<
  string,
  { given: unknown; then: unknown }
>;

interface Given {
  language: string;
  module: string;
  files: CodeText[];
  edges: Record<string, string[]>;
}

void fixture;

test("Import Gaps example 1: typescript relative imports", () => {
  const ex = EX["Import Gaps 1"] as { given: unknown; then: unknown };
  const g = ex.given as Given;
  const then = ex.then as LayerGap[];

  expect(importGaps(g.language, g.module, g.files, g.edges)).toStrictEqual(then);
});
