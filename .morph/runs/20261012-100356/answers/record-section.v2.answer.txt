import { readFileSync, statSync } from "node:fs";
import { join, posix } from "node:path";
import { loadContour, loadMap } from "../contour/load.js";
import { layerEdges } from "../rules/layers.js";
import { guardContract } from "../rules/guardrails.js";
import type { LanguageProfile } from "../language/types.js";
import { recordGaps, codeFiles } from "./recordGaps.js";
import { importGaps } from "./importGaps.js";
import type { CodeText } from "./importGaps.js";
import { renderRecord } from "./renderRecord.js";
import type { RecordSection } from "./renderRecord.js";

type Layers = Extract<RecordSection, { state: "read" }>["layers"];

function readText(root: string, rel: string): string | null {
  const p = join(root, rel);
  try {
    if (!statSync(p).isFile()) return null;
    return readFileSync(p, "utf8");
  } catch {
    return null;
  }
}

function firstLine(message: string): string {
  const i = message.indexOf("\n");
  return i === -1 ? message : message.slice(0, i);
}

function reasonFromError(error: string): string {
  const line = firstLine(error);
  if (line.endsWith(":")) return line.slice(0, line.length - 1);
  return line;
}

function goModule(text: string | null): string {
  if (text === null) return "";
  for (const line of text.split(/\r?\n/)) {
    const trimmed = line.trim();
    if (trimmed.startsWith("module ")) {
      return trimmed.slice("module ".length).trim();
    }
  }
  return "";
}

export function recordSection(
  root: string,
  files: readonly string[],
  profile: LanguageProfile,
  owned: readonly string[],
): string {
  const recordText = readText(root, "contour.yaml");
  if (recordText === null) {
    return renderRecord({ state: "not read", reason: "no contour.yaml" });
  }

  const mapText = readText(root, "morph-map.json");
  if (mapText === null) {
    return renderRecord({ state: "not read", reason: "no morph-map.json" });
  }

  const loadedRecord = loadContour(recordText, "contour.yaml");
  if (!loadedRecord.ok) {
    return renderRecord({
      state: "not read",
      reason: reasonFromError(loadedRecord.error),
    });
  }

  const loadedMap = loadMap(mapText, "morph-map.json");
  if (!loadedMap.ok) {
    return renderRecord({
      state: "not read",
      reason: reasonFromError(loadedMap.error),
    });
  }

  const record = loadedRecord.record;
  const map = loadedMap.map;

  const gaps = recordGaps({
    record,
    map,
    recordText,
    loaderWarnings: loadedRecord.warnings ?? [],
    files,
    profile,
    owned,
  });

  const contract = guardContract(readText(root, "decks/tools/guard.mjs") ?? "");

  let layers: Layers;
  if (contract < 2) {
    layers = {
      checked: false,
      reason: "decks/tools/guard.mjs speaks contract " + String(contract),
    };
  } else if (profile.id !== "typescript" && profile.id !== "go") {
    layers = { checked: false, reason: "no import rule for " + profile.id };
  } else {
    const edgesResult = layerEdges(record, map);
    if (!edgesResult.ok) {
      layers = { checked: false, reason: edgesResult.errors.join("; ") };
    } else {
      const edges = edgesResult.edges;
      const module = profile.id === "go" ? goModule(readText(root, "go.mod")) : "";
      const texts: CodeText[] = [];
      for (const file of codeFiles(profile, files)) {
        const dir = posix.dirname(file);
        if (!Object.hasOwn(edges, dir)) continue;
        const text = readText(root, file);
        if (text === null) continue;
        texts.push({ path: file, text });
      }
      layers = {
        checked: true,
        gaps: importGaps(profile.id, module, texts, edges),
      };
    }
  }

  return renderRecord({ state: "read", gaps, layers });
}
