import { cpSync, rmSync, writeFileSync } from "node:fs";
import { expect, test } from "vitest";
import { GO, PYTHON, TYPESCRIPT } from "../../src/language/profiles.js";
import { recordSection } from "../../src/primerRecord/recordSection.js";
import { fixture, fixtureJson, fixturePath, tmpRoot } from "../helpers.js";

const EX = fixtureJson("primerRecord/examples.json") as Record<
  string,
  { given: unknown; then: unknown }
>;

const REPO = fixturePath("primerRecord/repo");
const E1 = EX["Record Section 1"].given as { files: string[]; owned: string[] };

test("Record Section example 1: the fixture repo against its record", () => {
  expect(recordSection(REPO, E1.files, TYPESCRIPT, E1.owned)).toBe(
    fixture("primerRecord/section1.md"),
  );
});

test("Record Section example 2: a missing or invalid record file", () => {
  const then = EX["Record Section 2"].then as string[];
  const cases: Array<(t: ReturnType<typeof tmpRoot>) => void> = [
    () => {},
    (t) => {
      t.write("contour.yaml", fixture("primerRecord/repo/contour.yaml"));
    },
    (t) => {
      t.write("contour.yaml", "- a\n");
      t.write("morph-map.json", "{}");
    },
    (t) => {
      t.write("contour.yaml", fixture("primerRecord/repo/contour.yaml"));
      t.write("morph-map.json", '{"cards": 5}');
    },
  ];
  for (let i = 0; i < cases.length; i++) {
    const t = tmpRoot();
    try {
      cases[i](t);
      expect(recordSection(t.root, [], TYPESCRIPT, [])).toBe(then[i]);
    } finally {
      t.rm();
    }
  }
});

test("Record Section example 3: a tree whose guard speaks no contract", () => {
  const then = EX["Record Section 3"].then as string[];
  const t = tmpRoot();
  try {
    cpSync(REPO, t.root, { recursive: true });
    writeFileSync(t.path("decks/tools/guard.mjs"), "// the old guard\n");
    expect(recordSection(t.root, E1.files, TYPESCRIPT, E1.owned)).toBe(then[0]);
    rmSync(t.path("decks/tools/guard.mjs"));
    expect(recordSection(t.root, E1.files, TYPESCRIPT, E1.owned)).toBe(then[1]);
  } finally {
    t.rm();
  }
  expect(recordSection(REPO, E1.files, PYTHON, E1.owned)).toBe(then[2]);
});

test("Record Section example 4: a Go tree and its record", () => {
  const given = EX["Record Section 4"].given as {
    files: Record<string, string>;
    listed: string[];
    owned: string[];
    profile: string;
  };
  const then = EX["Record Section 4"].then as string;
  const t = tmpRoot();
  try {
    for (const [rel, text] of Object.entries(given.files)) {
      t.write(rel, text);
    }
    expect(recordSection(t.root, given.listed, GO, given.owned)).toBe(then);
  } finally {
    t.rm();
  }
});
