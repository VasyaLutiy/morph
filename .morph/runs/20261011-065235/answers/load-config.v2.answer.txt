```typescript
import fs from "node:fs";
import path from "node:path";
import { parseDocument } from "../contour/load.js";
import { errorDocument } from "./document.js";
import type { CommandResult } from "./types.js";
import { DEFAULT_SCOUT_SETTINGS } from "../scout/scoutCommand.js";
import type { ScoutSettings } from "../scout/scoutCommand.js";

export const CONFIG_FILE = "morph.config.yaml";

export interface ProjectConfig {
  version: 1;
  scout: ScoutSettings;
}

export type ConfigResult =
  | { ok: true; config: ProjectConfig }
  | { ok: false; problems: string[] };

export const DEFAULT_CONFIG: ProjectConfig = {
  version: 1,
  scout: DEFAULT_SCOUT_SETTINGS,
};

function isPlainObject(v: unknown): v is Record<string, unknown> {
  return typeof v === "object" && v !== null && !Array.isArray(v);
}

function isPositiveInteger(v: unknown): v is number {
  return typeof v === "number" && Number.isInteger(v) && v >= 1;
}

function collectUnknown(
  obj: Record<string, unknown>,
  pathName: string,
  known: readonly string[],
  problems: string[],
): void {
  const knownSet = new Set(known);
  for (const key of Object.keys(obj)) {
    if (!knownSet.has(key)) {
      problems.push(pathName + ": unknown key '" + key + "' (known: " + known.join(", ") + ")");
    }
  }
}

export function validateConfig(doc: unknown): ConfigResult {
  if (!isPlainObject(doc)) {
    return { ok: false, problems: ["(root): config must be an object"] };
  }

  const problems: string[] = [];

  collectUnknown(doc, "(root)", ["version", "scout"], problems);

  if (doc.version !== undefined && doc.version !== 1) {
    problems.push("version: must be 1");
  }

  let scoutInput: Record<string, unknown> | null = null;
  if (doc.scout !== undefined) {
    if (!isPlainObject(doc.scout)) {
      problems.push("scout: must be an object");
    } else {
      scoutInput = doc.scout;
    }
  }

  let grepSkip: string[] | null = null;
  let budgetValues: Record<string, number> | null = null;
  let capValues: Record<string, number> | null = null;
  let seedValues: Record<string, number> | null = null;

  if (scoutInput !== null) {
    collectUnknown(scoutInput, "scout", ["grepSkip", "budgets", "caps", "seed"], problems);

    const gs = scoutInput.grepSkip;
    if (gs !== undefined) {
      if (!Array.isArray(gs)) {
        problems.push("scout.grepSkip: must be a list");
      } else {
        const arr: readonly unknown[] = gs;
        const list: string[] = [];
        let invalid = false;
        for (let j = 0; j < arr.length; j++) {
          const item = arr[j];
          if (typeof item !== "string" || item === "") {
            problems.push("scout.grepSkip[" + j + "]: must be a non-empty string");
            invalid = true;
          } else {
            list.push(item);
          }
        }
        if (!invalid) {
          grepSkip = list;
        }
      }
    }

    const b = scoutInput.budgets;
    if (b !== undefined) {
      if (!isPlainObject(b)) {
        problems.push("scout.budgets: must be an object");
      } else {
        const known = ["calls", "reads", "chars", "rounds", "deadlineSeconds"];
        collectUnknown(b, "scout.budgets", known, problems);
        const values: Record<string, number> = {};
        let invalid = false;
        for (const key of known) {
          const v = b[key];
          if (v === undefined) continue;
          if (!isPositiveInteger(v)) {
            problems.push("scout.budgets." + key + ": must be a positive integer");
            invalid = true;
          } else {
            values[key] = v;
          }
        }
        if (!invalid) {
          budgetValues = values;
        }
      }
    }

    const c = scoutInput.caps;
    if (c !== undefined) {
      if (!isPlainObject(c)) {
        problems.push("scout.caps: must be an object");
      } else {
        const known = ["readLines", "readChars", "grepHits", "grepLineChars", "listEntries"];
        collectUnknown(c, "scout.caps", known, problems);
        const values: Record<string, number> = {};
        let invalid = false;
        for (const key of known) {
          const v = c[key];
          if (v === undefined) continue;
          if (!isPositiveInteger(v)) {
            problems.push("scout.caps." + key + ": must be a positive integer");
            invalid = true;
          } else {
            values[key] = v;
          }
        }
        if (!invalid) {
          capValues = values;
        }
      }
    }

    const s = scoutInput.seed;
    if (s !== undefined) {
      if (!isPlainObject(s)) {
        problems.push("scout.seed: must be an object");
      } else {
        const known = ["files", "chars"];
        collectUnknown(s, "scout.seed", known, problems);
        const values: Record<string, number> = {};
        let invalid = false;
        for (const key of known) {
          const v = s[key];
          if (v === undefined) continue;
          if (!isPositiveInteger(v)) {
            problems.push("scout.seed." + key + ": must be a positive integer");
            invalid = true;
          } else {
            values[key] = v;
          }
        }
        if (!invalid) {
          seedValues = values;
        }
      }
    }
  }

  if (problems.length > 0) {
    return { ok: false, problems };
  }

  const scout: ScoutSettings = {
    grepSkip: grepSkip !== null ? grepSkip : DEFAULT_CONFIG.scout.grepSkip.slice(),
    budgets: {
      calls: budgetValues?.calls ?? DEFAULT_CONFIG.scout.budgets.calls,
      reads: budgetValues?.reads ?? DEFAULT_CONFIG.scout.budgets.reads,
      chars: budgetValues?.chars ?? DEFAULT_CONFIG.scout.budgets.chars,
      rounds: budgetValues?.rounds ?? DEFAULT_CONFIG.scout.budgets.rounds,
      deadlineSeconds:
        budgetValues?.deadlineSeconds ?? DEFAULT_CONFIG.scout.budgets.deadlineSeconds,
    },
    caps: {
      readLines: capValues?.readLines ?? DEFAULT_CONFIG.scout.caps.readLines,
      readChars: capValues?.readChars ?? DEFAULT_CONFIG.scout.caps.readChars,
      grepHits: capValues?.grepHits ?? DEFAULT_CONFIG.scout.caps.grepHits,
      grepLineChars: capValues?.grepLineChars ?? DEFAULT_CONFIG.scout.caps.grepLineChars,
      listEntries: capValues?.listEntries ?? DEFAULT_CONFIG.scout.caps.listEntries,
    },
    seed: {
      files: seedValues?.files ?? DEFAULT_CONFIG.scout.seed.files,
      chars: seedValues?.chars ?? DEFAULT_CONFIG.scout.seed.chars,
    },
  };

  return { ok: true, config: { version: 1, scout } };
}

export function loadConfig(text: string | null, name: string = CONFIG_FILE): ConfigResult {
  if (text === null || text.trim() === "") {
    return { ok: true, config: DEFAULT_CONFIG };
  }
  const parsed = parseDocument(text, name);
  if (!parsed.ok) {
    return { ok: false, problems: [parsed.error] };
  }
  return validateConfig(parsed.doc);
}

export function readProjectConfig(
  root: string,
): { ok: true; config: ProjectConfig } | { ok: false; result: CommandResult } {
  const filePath = path.join(root, CONFIG_FILE);
  const stat = fs.statSync(filePath, { throwIfNoEntry: false });
  const text = stat?.isFile() ? fs.readFileSync(filePath, "utf8") : null;
  const loaded = loadConfig(text);
  if (loaded.ok) {
    return { ok: true, config: loaded.config };
  }
  const n = loaded.problems.length;
  const message =
    CONFIG_FILE +
    " is not a valid config (" +
    n +
    " problem" +
    (n === 1 ? "" : "s") +
    "): " +
    loaded.problems.join("; ");
  return {
    ok: false,
    result: {
      code: 2,
      document: errorDocument(2, "DeckError", message),
    },
  };
}
```
