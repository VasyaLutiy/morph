import type { Card } from "./types.js";

export const TRANSACTION_MARK = "# morph: subset transaction";

export function isTransactionDeck(cards: readonly Card[]): boolean {
  const prefix = TRANSACTION_MARK + "\n";
  for (const card of cards) {
    const acceptance = card.acceptance;
    if (typeof acceptance === "string" && acceptance.startsWith(prefix)) return true;
  }
  return false;
}

export interface Blame {
  cards: string[];
  outside: string[];
}

export function blameLog(
  log: string,
  card: string,
  owners: Record<string, string>,
  fileLines: readonly string[],
  exists: (path: string) => boolean,
): Blame {
  const blamed = new Set<string>();
  const outside = new Set<string>();
  let matchedAny = false;

  for (const raw of log.split("\n")) {
    const line = raw.trimEnd();
    if (line === "") continue;
    const first = line.charAt(0);
    if (first === "#" || first === " " || first === "\t") continue;
    for (const source of fileLines) {
      const m = new RegExp(source).exec(line);
      if (m === null) continue;
      const p = m[1];
      if (p === undefined) break;
      matchedAny = true;
      const rest = m[2];
      if (Object.prototype.hasOwnProperty.call(owners, p)) {
        blamed.add(owners[p] as string);
      } else if (!exists(p)) {
        blamed.add(card);
      } else {
        outside.add(p + (rest ?? ""));
      }
      break;
    }
  }

  if (!matchedAny) blamed.add(card);
  return { cards: Array.from(blamed).sort(), outside: Array.from(outside).sort() };
}
