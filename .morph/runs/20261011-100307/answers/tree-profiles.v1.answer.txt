    config: null,
    command: null,

    config: "full.json",
    command: "go build -overlay {config} ./... ; go vet -overlay {config} ./...",
  },
];

export function treeProfileFor(id: string): TreeProfile | null {
  for (const profile of TREE_PROFILES) {
    if (profile.id === id) {
      return profile;
    }
  }
  return null;
}
