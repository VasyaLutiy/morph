import fs from "node:fs";
import { test, expect } from "vitest";
import { runCommand } from "../../src/cli/runCommand.js";
import type { CliDeps, RunArgs, RunDocument } from "../../src/cli/types.js";
import { fixtureJson, tmpRepo, tmpRoot } from "../helpers.js";
import type { TmpRepo, TmpRoot } from "../helpers.js";

interface SuiteNetFixture {
  files: Record<string, string>;
  vitest: Record<string, string>;
  repo: Record<string, string>;
}

interface CardSpec {
  customId: string;
  target: string;
  acceptance: string;
  dependsOn?: string[];
}

interface RunSetup {
  r: TmpRepo;
  side: TmpRoot;
  deps: CliDeps;
}

function card(spec: CardSpec): Record<string, unknown> {
  const out: Record<string, unknown> = {
    customId: spec.customId,
    intent: "generate",
    targets: [spec.target],
    instruction: "x",
    acceptance: spec.acceptance,
  };
  if (spec.dependsOn !== undefined) {
    out.dependsOn = spec.dependsOn;
  }
  return out;
}

function setupRun(deck: unknown[], answers: readonly string[]): RunSetup {
  const r = tmpRepo();
  const side = tmpRoot();
  for (const id of answers) {
    side.write("answers/" + id + ".md", "```ts\nexport const " + id + " = 1;\n```\n");
  }
  side.write("deck.json", JSON.stringify(deck));
  const deps: CliDeps = {
    env: {
      PATH: process.env.PATH ?? "",
      HOME: r.root,
      GIT_CONFIG_NOSYSTEM: "1",
      GIT_AUTHOR_NAME: "Ada",
      GIT_AUTHOR_EMAIL: "ada@example.invalid",
      GIT_COMMITTER_NAME: "Ada",
      GIT_COMMITTER_EMAIL: "ada@example.invalid",
      MORPH_PROCESSOR_s_TYPE: "stub",
      MORPH_PROCESSOR_s_ANSWERS_DIR: side.path("answers"),
    },
    now: () => 1791310149000,
    cwd: r.root,
    transport: null,
  };
  return { r, side, deps };
}

function makeRunArgs(side: TmpRoot, runId: string): RunArgs {
  return {
    name: "run",
    root: ".",
    pretty: false,
    deck: side.path("deck.json"),
    processor: "s",
    runId,
    deadlineSeconds: 2400,
    maxCards: null,
    maxRetryBatches: 0,
  };
}

function scriptText(suite: SuiteNetFixture, name: string): string {
  const text = suite.vitest[name];
  if (text === undefined) {
    throw new Error("missing vitest script: " + name);
  }
  return text;
}

function repoText(suite: SuiteNetFixture, name: string): string {
  const text = suite.repo[name];
  if (text === undefined) {
    throw new Error("missing repo file: " + name);
  }
  return text;
}

function installVitest(r: TmpRepo, suite: SuiteNetFixture, script: string): void {
  fs.chmodSync(r.write("node_modules/.bin/vitest", scriptText(suite, script)), 0o755);
  r.write("tests/a.test.ts", repoText(suite, "tests/a.test.ts"));
  r.git(["add", "node_modules/.bin/vitest", "tests/a.test.ts"]);
  r.git(["commit", "-q", "-m", "net"]);
}

test("Run Command example 13: a red suite net after the run is code 1 and the suite key", async () => {
  const suite = fixtureJson("cli/suiteNet.json") as SuiteNetFixture;
  const s = setupRun(
    [
      card({
        customId: "a",
        target: "out/a.ts",
        acceptance:
          "test -f out/a.ts\necho '== full'; node_modules/.bin/vitest related --run out/a.ts",
      }),
      card({
        customId: "b",
        target: "out/b.ts",
        acceptance: "test -f out/b.ts",
        dependsOn: ["a"],
      }),
    ],
    ["a", "b"]
  );
  try {
    installVitest(s.r, suite, "runRed");
    const got = await runCommand(s.r.root, makeRunArgs(s.side, "r13"), s.deps);
    expect(got.code).toBe(1);
    const doc = got.document as RunDocument;
    expect(doc.runId).toBe("r13");
    expect(doc.report.outcomes.map((o) => o.customId + ":" + o.status)).toStrictEqual([
      "a:written",
      "b:written",
    ]);
    expect(doc.archive.ok).toBe(true);
    expect(doc.suite).toStrictEqual({
      command: "node_modules/.bin/vitest run --reporter=dot",
      ok: false,
      failing: ["tests/a.test.ts > A example 1: red"],
      cards: ["a"],
      log: " FAIL  tests/a.test.ts > A example 1: red\n",
    });
  } finally {
    s.r.rm();
    s.side.rm();
  }
});

test("Run Command example 14: a green suite net keeps the run at code 0 and the suite key last", async () => {
  const suite = fixtureJson("cli/suiteNet.json") as SuiteNetFixture;
  const s = setupRun(
    [
      card({
        customId: "a",
        target: "out/a.ts",
        acceptance:
          "test -f out/a.ts\necho '== full'; node_modules/.bin/vitest related --run out/a.ts",
      }),
      card({
        customId: "b",
        target: "out/b.ts",
        acceptance: "test -f out/b.ts",
        dependsOn: ["a"],
      }),
    ],
    ["a", "b"]
  );
  try {
    installVitest(s.r, suite, "runGreen");
    const got = await runCommand(s.r.root, makeRunArgs(s.side, "r14"), s.deps);
    expect(got.code).toBe(0);
    const doc = got.document as RunDocument;
    expect(doc.runId).toBe("r14");
    expect(doc.report.outcomes.map((o) => o.customId + ":" + o.status)).toStrictEqual([
      "a:written",
      "b:written",
    ]);
    expect(doc.suite).toStrictEqual({
      command: "node_modules/.bin/vitest run --reporter=dot",
      ok: true,
      failing: [],
      cards: [],
      log: "",
    });
    expect(Object.keys(doc)).toStrictEqual([
      "runId",
      "branch",
      "base",
      "report",
      "archive",
      "suite",
    ]);
  } finally {
    s.r.rm();
    s.side.rm();
  }
});

test("Run Command example 15: acceptances running no related tests leave the document with no suite key", async () => {
  const s = setupRun(
    [
      card({
        customId: "a",
        target: "out/a.ts",
        acceptance: "test -f out/a.ts",
      }),
      card({
        customId: "b",
        target: "out/b.ts",
        acceptance: "test -f out/b.ts",
        dependsOn: ["a"],
      }),
    ],
    ["a", "b"]
  );
  try {
    const got = await runCommand(s.r.root, makeRunArgs(s.side, "r15"), s.deps);
    expect(got.code).toBe(0);
    const doc = got.document as RunDocument;
    expect(doc.runId).toBe("r15");
    expect(doc.report.outcomes.map((o) => o.customId + ":" + o.status)).toStrictEqual([
      "a:written",
      "b:written",
    ]);
    expect(doc.suite).toBeUndefined();
    expect(Object.keys(doc)).toStrictEqual([
      "runId",
      "branch",
      "base",
      "report",
      "archive",
    ]);
  } finally {
    s.r.rm();
    s.side.rm();
  }
});
