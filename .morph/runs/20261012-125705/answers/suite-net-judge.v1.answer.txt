import fs from "node:fs";
import { describe, expect, test } from "vitest";
import { suiteNet } from "../../src/cli/suiteNet.js";
import { fixtureJson, tmpRoot } from "../helpers.js";
import type { Card } from "../../src/cards/types.js";

interface SuiteFixture {
  files: Record<string, string>;
  cards: { customId: string; targets: string[]; acceptance: string }[];
  vitest: Record<string, string>;
}

function vitestScript(f: SuiteFixture, key: string): string {
  const text = f.vitest[key];
  if (text === undefined) {
    throw new Error("suiteNet fixture has no vitest script named " + key);
  }
  return text;
}

function toCards(f: SuiteFixture): Card[] {
  return f.cards.map(
    (c) =>
      ({
        ...c,
        intent: "generate",
        contextSlice: [],
        instruction: "x",
        model: null,
        maxTokens: null,
        reasoning: null,
        variants: 1,
        dependsOn: [],
      }) as Card,
  );
}

function stage(t: ReturnType<typeof tmpRoot>, f: SuiteFixture, key: string): Card[] {
  for (const [rel, text] of Object.entries(f.files)) {
    t.write(rel, text);
  }
  fs.chmodSync(t.write("node_modules/.bin/vitest", vitestScript(f, key)), 0o755);
  return toCards(f);
}

const ENV: Record<string, string> = { PATH: process.env.PATH ?? "" };

describe("Suite Net", () => {
  test("Suite Net example 1: runs the whole suite, dedupes FAIL lines and maps files to cards", async () => {
    const f = fixtureJson("cli/suiteNet.json") as SuiteFixture;
    const t = tmpRoot();
    try {
      const cards = stage(t, f, "red");
      const got = await suiteNet(t.root, cards, ENV);
      expect(got).toStrictEqual({
        command: "node_modules/.bin/vitest run --reporter=dot",
        ok: false,
        failing: [
          "tests/a.test.ts > A example 1: red",
          "tests/c.test.ts [ tests/c.test.ts ]",
        ],
        cards: ["a", "c", "j"],
        log:
          " FAIL  tests/a.test.ts > A example 1: red\n" +
          " FAIL  tests/a.test.ts > A example 1: red\n" +
          " FAIL  tests/c.test.ts [ tests/c.test.ts ]\n",
      });
      expect(t.read("vitest.args")).toBe("run --reporter=dot\n");
    } finally {
      t.rm();
    }
  });

  test("Suite Net example 2: a green suite reports ok with no failing files", async () => {
    const f = fixtureJson("cli/suiteNet.json") as SuiteFixture;
    const t = tmpRoot();
    try {
      const cards = stage(t, f, "green");
      const got = await suiteNet(t.root, cards, ENV);
      expect(got).toStrictEqual({
        command: "node_modules/.bin/vitest run --reporter=dot",
        ok: true,
        failing: [],
        cards: [],
        log: "",
      });
      expect(t.read("vitest.args")).toBe("run --reporter=dot\n");
    } finally {
      t.rm();
    }
  });

  test("Suite Net example 3: no card asks for the suite, so nothing runs", async () => {
    const f = fixtureJson("cli/suiteNet.json") as SuiteFixture;
    const t = tmpRoot();
    try {
      for (const [rel, text] of Object.entries(f.files)) {
        t.write(rel, text);
      }
      fs.chmodSync(t.write("node_modules/.bin/vitest", vitestScript(f, "red")), 0o755);
      const cards = toCards(f).map((c) => ({ ...c, acceptance: "exit 0" }) as Card);
      const got = await suiteNet(t.root, cards, ENV);
      expect(got).toBe(null);
      expect(t.exists("vitest.args")).toBe(false);
    } finally {
      t.rm();
    }
  });

  test("Suite Net example 4: a broken run with no FAIL line yields no failing files", async () => {
    const f = fixtureJson("cli/suiteNet.json") as SuiteFixture;
    const t = tmpRoot();
    try {
      const cards = stage(t, f, "broken");
      const got = await suiteNet(t.root, cards, ENV);
      expect(got).toStrictEqual({
        command: "node_modules/.bin/vitest run --reporter=dot",
        ok: false,
        failing: [],
        cards: [],
        log:
          "Error: [vitest-worker]: Timeout calling onTaskUpdate\n" +
          " Tests  3 passed (3)\n",
      });
    } finally {
      t.rm();
    }
  });
});
