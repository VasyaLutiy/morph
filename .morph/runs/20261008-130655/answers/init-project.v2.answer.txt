import * as fs from "node:fs";
import * as path from "node:path";
import { fileURLToPath } from "node:url";

export type InitLanguage = "typescript" | "python" | "go";

export interface InitOptions {
  project: string;
  language: InitLanguage;
  module: string | null;
  templates: string | null;
}

export interface InitValues {
  name: string;
  module: string;
  language: string;
}

export interface InitDocument {
  name: string;
  language: InitLanguage;
  module: string;
  files: string[];
}

export interface InitResult {
  code: 0 | 2;
  document: unknown;
}

export const COMMON_DIR = "common";

const PLACEHOLDER_RE = /\{\{(name|module|language)\}\}/g;
const MODE_MASK = 0o777;

export function defaultTemplatesDir(): string {
  return path.resolve(path.dirname(fileURLToPath(import.meta.url)), "..", "..", "templates");
}

export function fillTemplate(text: string, values: InitValues): string {
  return text.replace(PLACEHOLDER_RE, (match: string, key: string): string => {
    if (key === "name") return values.name;
    if (key === "module") return values.module;
    if (key === "language") return values.language;
    return match;
  });
}

export function listTemplateFiles(dir: string): string[] {
  if (!isDirectory(dir)) return [];
  const found: string[] = [];
  collectFiles(dir, "", found);
  found.sort(byCodeUnit);
  return found;
}

export function initProject(root: string, args: InitOptions, cwd: string): InitResult {
  const templates = args.templates === null ? defaultTemplatesDir() : path.resolve(cwd, args.templates);

  const target = statOrNull(root);
  if (target !== null && !target.isDirectory()) {
    return refusal("target is not a directory");
  }
  if (target !== null) {
    const entries = fs.readdirSync(root);
    if (entries.length > 0) {
      const first = [...entries].sort(byCodeUnit)[0] as string;
      return refusal(`target directory is not empty (first entry: ${first})`);
    }
  }

  const commonDir = path.join(templates, COMMON_DIR);
  if (!isDirectory(commonDir)) {
    return refusal(`template folder not found: ${COMMON_DIR}`);
  }
  const languageDir = path.join(templates, args.language);
  if (!isDirectory(languageDir)) {
    return refusal(`template folder not found: ${args.language}`);
  }

  const sources = new Map<string, string>();
  for (const relative of listTemplateFiles(commonDir)) {
    sources.set(relative, path.join(commonDir, relative));
  }
  for (const relative of listTemplateFiles(languageDir)) {
    sources.set(relative, path.join(languageDir, relative));
  }
  const files = [...sources.keys()].sort(byCodeUnit);

  const values: InitValues = {
    name: args.project,
    module: args.module ?? args.project,
    language: args.language,
  };

  fs.mkdirSync(root, { recursive: true });
  for (const relative of files) {
    const source = sources.get(relative) as string;
    const destination = path.join(root, relative);
    fs.mkdirSync(path.dirname(destination), { recursive: true });
    fs.writeFileSync(destination, fillTemplate(fs.readFileSync(source, "utf8"), values));
    fs.chmodSync(destination, fs.statSync(source).mode & MODE_MASK);
  }

  const document: InitDocument = {
    name: args.project,
    language: args.language,
    module: values.module,
    files,
  };
  return { code: 0, document };
}

function statOrNull(target: string): fs.Stats | null {
  try {
    return fs.statSync(target);
  } catch {
    return null;
  }
}

function isDirectory(dir: string): boolean {
  const stat = statOrNull(dir);
  return stat !== null && stat.isDirectory();
}

function collectFiles(dir: string, prefix: string, out: string[]): void {
  const entries = fs.readdirSync(dir, { withFileTypes: true });
  for (const entry of entries) {
    const relative = prefix === "" ? entry.name : `${prefix}/${entry.name}`;
    if (entry.isDirectory()) {
      collectFiles(path.join(dir, entry.name), relative, out);
    } else if (entry.isFile()) {
      out.push(relative);
    }
  }
}

function byCodeUnit(left: string, right: string): number {
  if (left < right) return -1;
  if (left > right) return 1;
  return 0;
}

function refusal(message: string): InitResult {
  return { code: 2, document: { error: { code: 2, kind: "RefusalError", message } } };
}
